\vertex (i2) at (-1,-1) {$q'$};

\node[
    circle,
    fill=black,
    minimum size=5mm,
    inner sep=0pt

\vertex (e) at (2,0.5);
\vertex (f) at (2,-0.5);

\vertex (l1) at (3,0.8) {$l_1$};
\vertex (l2) at (3,0.3) {$l_2$};

\vertex (l4) at (3,-0.8) {$l_4$};

  (i2) -- [fermion] (b),
  (a) -- [fermion] (f1),
  (b) -- [fermion] (f2),
  
  (a) -- [boson, edge label'=\(V\)] (v),
  (b) -- [boson, edge label'=\(V\)] (v),

  (e) -- [fermion] (l2),
  (f) -- [fermion] (l3),

};

\end{feynman}
\end{tikzpicture}
